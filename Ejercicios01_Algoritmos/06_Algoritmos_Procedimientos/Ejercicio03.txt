// Ingrese por teclado el nombre, el numero de horas trabajadas durante la semana, y la tarifa por hora 
// esta informacion es de un obrero cuya jornada laboral es de 48 horas semanales. La política de pagos
// de la empresa consiste en que el valor de la hora extra tiene un incremento del 40%. Se deberá visualizar 
// solo del trabajador que hizo horas extras, el valor de la hora extra y el valor del importe total.
SubProceso imprimirHorasExtras(horastrabajadas,tarifaPorHoras,horasExtras, importeTotal,valorPorHorasExtras)
	horasExtras = horastrabajadas - 48
	valorPorHorasExtras = (horasExtras * tarifaPorHoras) * 0.4
	importeTotal = valorPorHorasExtras + (horastrabajadas  * tarifaPorHoras)
	Escribir "*************************************"
	Escribir "Empleado : " nombreTrabajador
	Escribir "Horas Trabajadas Semanales : " horastrabajadas " Hrs"
	Escribir "Horas Extras : " horasExtras " Hr"
	Escribir "Importe por horas trabajadas : S/ " tarifaPorHoras
	Escribir "Tarifa Total H.E. : S/ " valorPorHorasExtras
	Escribir "*************************************"
	Escribir "Total a pagar : 	S/ 	" importeTotal
FinSubProceso

SubProceso imprimirTotalMenor(nombreTrabajador,horasTrabajadas,tarifaPorhoras)
	Escribir "*************************************"
	Escribir "Empleado : " nombreTrabajador

	Escribir "*************************************"

FinSubProceso
