// Leer los coeficientes de una parábola f(x)= ax^2+bx+c y una recta g(x)= mx+n
// Si se intersectan en dos puntos reales distintos, 
// calcular el área encerrada entre ambas funciones.

Proceso Area_Parabola_Recta
	Definir valorA, valorB, valorC, m, n Como Real
	Definir A, B, C, delta, x1, x2, I, area Como Real
	Definir H1, H2 Como Real
	
	Escribir "Ingrese a, b, c de la parabola f(x)= ax^2 + bx + c : "
	Leer a, b, c
	
	Escribir "Ingrese m, n de la recta g(x) = mx + n : "
	Leer m, n
	
	A = valorA
	B = valorB - m
	C = valorC - n
	
	Si A <> 0 Entonces
		delta <- B^2 - 4*A*C
		
		Si delta > 0 Entonces
			x1 <- (-B - Rc(delta)) / (2*A)
			x2 <- (-B + Rc(delta)) / (2*A)
			
			H1 <- (A/3)*x1^3 + (B/2)*x1^2 + C*x1
			H2 <- (A/3)*x2^3 + (B/2)*x2^2 + C*x2
			
			I <- H2 - H1
			
			Si I < 0 Entonces
				area <- -I
			Sino
				area <- I
			FinSi
			
			Escribir "Los puntos de interseccion son:"
			Escribir "x1 = ", x1
			Escribir "x2 = ", x2
			Escribir "El area encerrada es: ", area
		Sino
			Si delta = 0 Entonces
				Escribir "Solo existe un punto de tangencia. No hay area encerrada."
			Sino
				Escribir "No existen dos intersecciones reales. No se puede formar area cerrada."
			FinSi
		FinSi
	Sino
		Si B <> 0 Entonces
			Escribir "La diferencia entre funciones es lineal, no una parabola."
			Escribir "No se forma una region cerrada entre ambas."
		Sino
			Escribir "Las funciones son paralelas o coincidentes."
		FinSi
	FinSi
FinProceso
