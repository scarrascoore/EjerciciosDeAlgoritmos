// si su nota es mayor o igual a 11; en caso contrario, está desaprobado. 
// Finalmente, mostrar el nombre, la nota y el estado del estudiante.

	Definir nota Como Real
	
	Escribir "Ingrese el nombre del estudiante:"
	Leer nombre
		
	Escribir "Ingrese la nota final:"
	Leer nota
		
	Si nota >= 11 Entonces
		Escribir "Estudiante: ", nombre
		Escribir "Nota final: ", nota
		Escribir "Estado: APROBADO"

		Escribir "Estudiante: ", nombre
		Escribir "Nota final: ", nota

FinProceso
